# Runtime dependencies for the API container only.
#
# The full requirements.txt covers three different jobs: serving the API,
# downloading documents, and building the database. Only the first runs in the
# deployed image, so installing all three pulls in playwright, pandas, pyarrow,
# and pdfplumber that no request path imports — hundreds of MB of image weight
# and build time for code that never executes in production.
#
# Kept honest by tests/test_web_group/test_runtime_requirements.py, which
# fails if api/ or utils/ grows an import that is not satisfiable from this
# file. If that test fails, either the import belongs behind a lazy import in
# the pipeline layer, or this file needs the dependency added deliberately.
#
# Use requirements.txt (not this file) for the pipeline, the downloader, and
# any local development that runs them.

fastapi>=0.109,<1.0
starlette>=0.35,<2.0     # transitive via fastapi, but pinned: 1.0 changed app.routes
uvicorn[standard]>=0.25,<1.0
jinja2>=3.1,<4           # HTML templates
python-multipart>=0.0.6,<1.0  # form data parsing

# api/routes/frontend.py imports pipeline.builder for EXHIBIT_TYPES, and that
# module imports openpyxl at import time — so the API cannot start without it.
openpyxl>=3.0,<4

# XLSX export from the Keyword Explorer (api/routes/keyword_xlsx.py).
xlsxwriter>=3.1,<4

# utils/ uses requests for outbound HTTP.
requests>=2.28,<3
